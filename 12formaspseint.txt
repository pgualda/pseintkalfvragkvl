Algoritmo manera1
	// estrategia:
	// super simple vamos nro x nro preguntando si es mayor que 
	// los otros dos
	escribir "Ingresa 3 enteros positivos distintos para"
	escribir "saber cual es el mayor, manera nro 1"
	leer a,b,c
	si a>b
		si a>c
			mayor=a
		FinSi
	FinSi
	si b>c
		si b>a
			mayor=b
		FinSi
	FinSi
	si c>a
		si c>b
			mayor=c
		FinSi
	FinSi
    escribir "el mayor es:",mayor	
FinAlgoritmo

Algoritmo manera9
	// estrategia:
	// cargar los valores en un arreglo y luego hacer una primer pasada
	// de un metodo de burbuja, el ultimo elemento sera el mayor.
	// muy facil de escalar y util para empezar a conocer el metodo de burbuja
	Dimensionar n[3]
	escribir "Ingresa 3 enteros positivos distintos para"
	escribir "saber cual es el mayor, manera nro 9"
	leer n[1],n[2],n[3]
	para x=1 hasta 2
		si n[x]>n[x+1]
			w=n[x]
			n[x]=n[x+1]
			n[x+1]=w
		FinSi
	FinPara
    escribir "el mayor es:",n[3]
FinAlgoritmo

Algoritmo manera12
	// estrategia:
	// con un poco de complejidad adicional vamos a llevar
	// 3 contadores a 0.el que mas tarda que sera el mayor.
	// al comenzar tendremos una mascara
	// en binario de todos 1 (7 en decimal, 4+2+1) y al irse
	// cerrando las cuentas regresivas iremos cambiando la
	// mascara para que solo reste en los contadores que
	// aun tienen cantidad.
	// estan comentadas varias lineas que permiten ver el estado de las variables.
	escribir "Ingresa 3 enteros positivos distintos para"
	escribir "saber cual es el mayor, manera nro 12"
	leer a,b,c
	// copiamos los valores
	ca=a
	cb=b
	cc=c
	mascara=7 // todos uno 4+2+1
	mientras mascara<>4 y mascara<>2 y mascara<>1 // 100 - 10 - 1 indican salida, solo queda un valor.
		// escribir "antes  :",a, " ", b, " ", c, " ", mascara
		a=a-obtienebit(mascara,0)
        si a<1 y obtienebit(mascara,0)=1
  			mascara=mascara-1
    	FinSi
		b=b-obtienebit(mascara,1)

  			mascara=mascara-2
    	FinSi

        si c<1 y obtienebit(mascara,2)=1
  			mascara=mascara-4
    	FinSi

		// esperar tecla
	FinMientras
